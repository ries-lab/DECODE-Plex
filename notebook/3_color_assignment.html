hist, bin_edges = np.histogram(phot_norm, bins=100, density=True)
bin_centers = (bin_edges[:-1] + bin_edges[1:]) / 2

peaks, _ = find_peaks(hist, distance=5, prominence=0.02)

ratios = bin_centers[peaks]
